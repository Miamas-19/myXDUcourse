library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity 2mux1 is
Port ( 
        y:out std_logic;
        a:in std_logic;
	b:in std_logic;
	s:in std_logic
        );
end 2mux1;

architecture Behavioral of 2mux1 is

begin
process(switch)
begin
	y<=a when s='0' else
		b when s='1';
	end process;

end Behavioral;
