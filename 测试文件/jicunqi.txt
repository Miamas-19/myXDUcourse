library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity jicunqi is
Port ( 
    y:out std_logic_vector(7 downto 0);
    x:in std_logic_vector(7 downto 0);
	lod:in std_logic;
	clk:in std_logic;
	l:in std_logic;
	r:in std_logic;
	reset:in std_logic
    );
end jicunqi;

architecture one of jicunqi is
    signal q:std_logic_vector(7 downto 0);
begin
    process(reset,lod,clk,l,r)
    begin
        if reset='0' then q<="00000000";
elsif clk'event and clk='1' then
            if (lod='1') then q<=x;
            elsif (r='1') then
                for i in 7 downto 1 loop q(i-1)<=q(i);
                end loop;
		q(7)<=r;
            else
                for i in 0 to 6 loop q(i+1)<=q(i);
                end loop;
		q(0)<=l;
            end if;
        end if;
        q<=q;
    end process;
end one;
