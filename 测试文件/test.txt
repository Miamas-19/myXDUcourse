library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity test is
Port ( 
        led:out std_logic;
        switch:in std_logic
        );
end test;

architecture Behavioral of test is

begin
process(switch)
begin
    if switch='1' then
    led<='1';
    else 
    led<='0';
    end if;
end process;

end Behavioral;
