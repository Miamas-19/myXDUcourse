# 线上实验——原码二位乘法器

## VHDL设计

```vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_unsigned.ALL;
use ieee.numeric_std.all;

entity mul is
    Port ( 
           start : in STD_LOGIC;
           clk : in STD_LOGIC;
           a : in STD_LOGIC_VECTOR(6 downto 0);
           b : in STD_LOGIC_VECTOR(6 downto 0);
           done : out STD_LOGIC;
           dout : out STD_LOGIC_VECTOR(12 downto 0));
end mul;

architecture Behavioral of mul is

begin
    process(clk,start)
    variable count : integer :=1;
    variable reg:std_logic_vector(15 downto 0);
    variable a1:std_logic_vector(8 downto 0);
    variable a2:std_logic_vector(8 downto 0);
    variable b1:std_logic_vector(5 downto 0);
    variable c:std_logic:='0';
    begin
        if(start='1') then
            reg:=(others=>'0');
            a1(8 downto 0) := "000" & a(5 downto 0);
            a2(8 downto 0) := "00" & a(5 downto 0) & '0';
            b1(5 downto 0) := b(5 downto 0);
            done<='0';
            reg:="000000000" & b1 &'0';
        else
            if(clk'event and clk='1') then
            if(count<4) then
                reg(0):=c;
                if (reg(2 downto 0)="000") then 
                    c:='0';
                elsif (reg(2 downto 0)="001" or reg(2 downto 0)="010") then 
                    reg(15 downto 7):=std_logic_vector(unsigned(reg(15 downto 7))+unsigned(a1(8 downto 0)));
                    c:='0';
                elsif (reg(2 downto 0)="011" or reg(2 downto 0)="100") then 
                    reg(15 downto 7):=std_logic_vector(unsigned(reg(15 downto 7))+unsigned(a2(8 downto 0)));
                    c:='0';
                elsif (reg(2 downto 0)="101" or reg(2 downto 0)="110") then 
                    reg(15 downto 7):=std_logic_vector(unsigned(reg(15 downto 7))+unsigned(not a1)+1);
                    c:='1';
                elsif (reg(2 downto 0)="111") then 
                    c:='1';
                end if;
                reg(14 downto 1):=reg(15 downto 2);
                reg(14 downto 1):=reg(15 downto 2);
                count:=count+1;
            else
            dout<=(a(6)xor b(6)) & reg(12 downto 1);
            done<='1';
            end if;
            end if;
        end if;
    end process;

end Behavioral;
```

## 激励文件

```vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity multb is
end multb;

architecture Behavioral of multb is
signal start:std_logic:='1';
signal clk:std_logic:='0';
signal done:std_logic:='0';
signal a:std_logic_vector(6 downto 0);
signal b:std_logic_vector(6 downto 0);
signal dout:std_logic_vector(12 downto 0);

component mul is
    Port ( 
       start : in STD_LOGIC;
       clk : in STD_LOGIC;
       a : in STD_LOGIC_VECTOR(6 downto 0);
       b : in STD_LOGIC_VECTOR(6 downto 0);
       done : out STD_LOGIC;
       dout : out STD_LOGIC_VECTOR(12 downto 0));
end component;

begin
    dut:mul port map(
    start=>start,
    clk=>clk,
    a=>a,
    b=>b,
    dout=>dout,
    done=>done
    );
    
    process(clk)
    begin
        clk <= not clk after 10ns;
    end process;
    
    process
    begin
    a<="0100111";
    b<="1100111";
    wait for 40ns;
    start<='0';
    wait;
    end process;

end Behavioral;
```
