# 线上实验——时钟模块

## VHDL设计

```vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_unsigned.all;

entity clk_gen is
    port(
    clk,reset:in std_logic;
    clk1,nclk1:out std_logic;
    clk2,nclk2:out std_logic;
    w0,w1,w2,w3:out std_logic);
end clk_gen;

architecture Behavioral of clk_gen is
signal count:std_logic_vector(3 downto 0):="0000";
signal q:std_logic_vector(2 downto 0):="000";
signal clk2_1:std_logic;
signal nclk2_1:std_logic;

begin
    process(clk,reset)
        begin
        if(reset='1') then
        clk1<=clk;
        nclk1<=not clk;
        if(clk'event and clk='1')then
            if(count="1111")then count<="0000";
            else count<=count+1;
            end if;
        end if;
        end if;
    end process;
    clk2<=count(0);
    clk2_1<=count(0);
    nclk2<=not clk2_1;
    
    process(clk,reset)
        begin
        if(reset='1')then
        
        if(clk'event and clk='1') then q<=q+1;
            if(q="000" or q="001")      then w0<='1';w1<='0';w2<='0';w3<='0';
            elsif(q="010" or q="011")   then w0<='0';w1<='1';w2<='0';w3<='0';
            elsif(q="100" or q="101")   then w0<='0';w1<='0';w2<='1';w3<='0';
            elsif(q="110" or q="111")   then w0<='0';w1<='0';w2<='0';w3<='1';
            end if;
        end if;
        end if;
    end process;

end Behavioral;
```

## 激励文件

```vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity clktb is
end clktb;

architecture Behavioral of clktb is
component clk_gen port(
    clk,reset:in std_logic;
    clk1,nclk1:out std_logic;
    clk2,nclk2:out std_logic;
    w0,w1,w2,w3:out std_logic);
end component;
signal clk:std_logic;
signal reset:std_logic;
signal clk1:std_logic;
signal clk2:std_logic;
signal nclk1:std_logic;
signal nclk2:std_logic;
signal w0:std_logic;
signal w1:std_logic;
signal w2:std_logic;
signal w3:std_logic;
begin
    dut:clk_gen port map(
    clk=>clk,
    reset=>reset,
    clk1=>clk1,
    nclk1=>nclk1,
    clk2=>clk2,
    nclk2=>nclk2,
    w0=>w0,
    w1=>w1,
    w2=>w2,
    w3=>w3
    );
    
    process
    begin
        clk<='1';
        wait for 10ns;
        clk<='0';
        wait for 10ns;
    end process;
    
    process
    begin
        reset<='0';
        wait for 20ns;
        reset<='1';
        wait for 200ns;
    end process;

end Behavioral;
```
