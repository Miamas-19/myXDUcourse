# 线上实验——布斯乘法器

## VHDL设计

```vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_unsigned.ALL;
use ieee.numeric_std.all;

entity booth is
    Port ( 
           start : in STD_LOGIC;
           clk : in STD_LOGIC;
           a : in STD_LOGIC_VECTOR(7 downto 0);
           b : in STD_LOGIC_VECTOR(7 downto 0);
           done : out STD_LOGIC;
           dout : out STD_LOGIC_VECTOR(15 downto 0));
end booth;

architecture Behavioral of booth is

begin
    process(clk,start)
    variable count : integer :=1;
    variable reg:std_logic_vector(17 downto 0);
    variable a1:std_logic_vector(8 downto 0);
    begin
        if(start='1') then
            reg:=(others=>'0');
            a1(8 downto 0) := a(7) & a(7 downto 0);
            done<='0';
            reg:="000000000" & b & "0";
        else
            if(clk'event and clk='1') then
            if(count<8) then
                if (reg(1 downto 0)="01") then reg(17 downto 9):=std_logic_vector(unsigned(reg(17 downto 9)) + unsigned(a1(8 downto 0)));
                elsif (reg(1 downto 0)="10") then reg(17 downto 9):=std_logic_vector(unsigned(reg(17 downto 9))+unsigned(not a1)+1);
                else  null;
                end if;
                reg(16 downto 0):=reg(17 downto 1);
                count:=count+1;
                --dout<=reg(17 downto 2);
            else
            dout<=reg(17 downto 2);
            done<='1';
            end if;
            end if;
        end if;
    end process;

end Behavioral;
```

## 激励文件

```vhdl
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity boothtb is
end boothtb;

architecture Behavioral of boothtb is
signal start:std_logic:='1';
signal clk:std_logic:='0';
signal done:std_logic:='0';
signal a:std_logic_vector(7 downto 0);
signal b:std_logic_vector(7 downto 0);
signal dout:std_logic_vector(15 downto 0);

component booth is
    Port ( 
       start : in STD_LOGIC;
       clk : in STD_LOGIC;
       a : in STD_LOGIC_VECTOR(7 downto 0);
       b : in STD_LOGIC_VECTOR(7 downto 0);
       done : out STD_LOGIC;
       dout : out STD_LOGIC_VECTOR(15 downto 0));
end component;

begin
    dut:booth port map(
    start=>start,
    clk=>clk,
    a=>a,
    b=>b,
    dout=>dout,
    done=>done
    );
    
    process(clk)
    begin
        clk <= not clk after 10ns;
    end process;
    
    process
    begin
    a<="00001010";
    b<="00001010";
    wait for 40ns;
    start<='0';
    wait;
    end process;

end Behavioral;
```
